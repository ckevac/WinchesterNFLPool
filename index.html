/** * THE ARENA - NFL PICK'EM BACKEND */ const PICKS_SHEET_NAME = "Picks"; const PLAYERS_SHEET_NAME = "Players"; const SURVIVOR_SHEET_NAME = "Survivor"; function doGet(e) { const ss = SpreadsheetApp.getActiveSpreadsheet(); const action = e.parameter.action; // 1. Fetch Player list for login if (action === "getPlayers") { const sheet = ss.getSheetByName(PLAYERS_SHEET_NAME); if (!sheet) return JSONResponse({ players: [] }); const names = sheet.getRange("A2:A100").getValues().flat().filter(n => n !== ""); return JSONResponse({ players: names }); } // Login now lives in doPost (see below) so the passcode is never sent as a URL // query parameter - it used to land in browser history and in this script's own // execution logs in plaintext. Intentionally no "login" branch here anymore. // 2. Get Survivor Data (Used for restricted teams, and the Survivor board) if (action === "getSurvivorData") { const sheet = ss.getSheetByName(SURVIVOR_SHEET_NAME); if (!sheet) return JSONResponse({ players: [] }); const lastRow = sheet.getLastRow(); if (lastRow < 2) return JSONResponse({ players: [] }); // Get headers and data const data = sheet.getRange(1, 1, lastRow, 22).getValues(); const players = []; // Start from index 1 to skip header row for (let i = 1; i < data.length; i++) { players.push({ name: data[i][0], // Column A ("Player") status: data[i][19], // Column T ("Alive / Eliminated") - was wrongly Column E // (which is actually the Week 4 pick), inherited from // the original script, not something introduced here. usedTeams: data[i][21] // Column V ("Teams Picked") }); } return JSONResponse({ players: players }); } // 3. Get Weekly/Overall pick accuracy for every player. // The Players sheet already has this pre-computed (Overall Accuracy Percentage, then // a Week N Accuracy Percentage / Week N Combined Score Differential pair per week) - // almost certainly formulas driven by the Picks sheet's "Right Pick?" column. Serve // that directly instead of re-deriving scoring logic here. Password (Column B) and // Admin (Column C) are intentionally never included in the response. if (action === "getLeaderboardData") { const sheet = ss.getSheetByName(PLAYERS_SHEET_NAME); if (!sheet) return JSONResponse({ players: [] }); const lastRow = sheet.getLastRow(); const lastCol = sheet.getLastColumn(); if (lastRow < 2) return JSONResponse({ players: [] }); const data = sheet.getRange(1, 1, lastRow, lastCol).getValues(); const players = []; for (let i = 1; i < data.length; i++) { const row = data[i]; if (!row[0]) continue; // skip blank rows const weekly = {}; // Column D (index 3) = Overall Accuracy %, then pairs starting at Column E // (index 4): Week N Accuracy % at index 4 + (N-1)*2, Week N Score Differential // right after it - the differential isn't used by the frontend today. for (let week = 1, col = 4; col < row.length; week++, col += 2) { const cell = row[col]; if (cell === "" || cell === null || cell === "#N/A") continue; weekly[week] = parsePercent_(cell); } players.push({ name: row[0], overall: parsePercent_(row[3]), weekly: weekly // e.g. { "1": 0.4375, "2": 0.4375, "3": 0.75, "4": 0.5625 } }); } return JSONResponse({ players: players }); } return JSONResponse({ error: "Invalid Action" }); } // Accuracy cells are formatted as percentages, so getValues() normally returns the raw // fraction already (e.g. 0.5469 for "54.69%"). Handles a literal-text cell defensively too. function parsePercent_(v) { if (v === "" || v === null || v === undefined || v === "#N/A") return 0; if (typeof v === "number") return v; const s = v.toString().replace("%", "").trim(); const n = parseFloat(s); if (isNaN(n)) return 0; return v.toString().indexOf("%") !== -1 ? n / 100 : n; } function doPost(e) { const ss = SpreadsheetApp.getActiveSpreadsheet(); try { const data = JSON.parse(e.postData.contents); // 1. Handle User Login (moved here from doGet - see note above) if (data.action === "login") { const name = (data.name || "").toString(); const pass = (data.pass || "").toString(); const sheet = ss.getSheetByName(PLAYERS_SHEET_NAME); if (!sheet) return JSONResponse({ success: false }); const rows = sheet.getRange("A2:C100").getValues(); for (let i = 0; i < rows.length; i++) { if (rows[i][0].toString().trim() === name.trim() && rows[i][1].toString() === pass) { return JSONResponse({ success: true, isAdmin: (rows[i][2] === "Admin" || rows[i][2] === true) }); } } return JSONResponse({ success: false }); } // 2. Handle Pick Submission if (data.action === "submitPicks") { const sheet = ss.getSheetByName(PICKS_SHEET_NAME) || ss.insertSheet(PICKS_SHEET_NAME); const timestamp = new Date(); const rows = sheet.getDataRange().getValues(); for (let i = rows.length - 1; i >= 1; i--) { if (rows[i][1] === data.user && rows[i][2] == data.week) { sheet.deleteRow(i + 1); } } for (let gameId in data.picks) { sheet.appendRow([ timestamp, data.user, data.week, gameId, data.picks[gameId], data.tieBreaker, data.survivorPick ]); } return JSONResponse({ success: true }); } return JSONResponse({ success: false, error: "Invalid Action" }); } catch (err) { return JSONResponse({ success: false, error: err.toString() }); } } function JSONResponse(obj) { return ContentService.createTextOutput(JSON.stringify(obj)) .setMimeType(ContentService.MimeType.JSON); }
