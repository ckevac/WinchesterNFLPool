Same rule results_sync.gs uses. function guessSeasonYear_() { const now = new Date(); return now.getMonth() <= 1 ? now.getFullYear() - 1 : now.getFullYear(); } // Shared by every ESPN call in this project (this file's getWeekScheduleInfo_, and // results_sync.gs's getCurrentSeasonInfo_ / syncWeek_ - all .gs files in one Apps // Script project share a single global scope, so this is callable from either file). // // Live testing (Week 1, actual game day) showed UrlFetchApp calls to ESPN's // undocumented scoreboard API coming back with an HTML page instead of JSON, every // single time, even though the exact same URL returned clean JSON from a normal // browser - a couple of quick retries didn't help, because it wasn't a one-off blip. // That's the standard signature of a site/CDN blocking or challenge-paging requests // that don't look like they came from a browser: Apps Script's default UrlFetchApp // request carries no User-Agent a WAF would recognize as one. On total failure, the thrown error // includes the real HTTP status and a snippet of the body, so if this ever fails // again the error text itself says what actually came back instead of just "not // valid JSON". function fetchEspnJson_(url) { const options = { muteHttpExceptions: true, headers: { 'User-Agent': 'Mozilla/5.0 (Windows NT 10.0; Win64; x64) AppleWebKit/537.36 (KHTML, like Gecko) Chrome/128.0.0.0 Safari/537.36', 'Accept': 'application/json' } }; let lastErr; for (let attempt = 0; attempt < 3; attempt++) { try { const resp = UrlFetchApp.fetch(url, options); const body = resp.getContentText(); try { return JSON.parse(body); } catch (parseErr) { throw new Error('ESPN returned HTTP ' + resp.getResponseCode() + ' (non-JSON): ' + body.slice(0, 200)); } } catch (e) { lastErr = e; if (attempt < 2) Utilities.sleep(400 * (attempt + 1)); } } throw lastErr; } // Fetches (and caches for 5 minutes) a week's schedule from ESPN: Game ID, kickoff // time, and home/away team for every game. only // Apps Script's own server IPs are. const sheet = ss.getSheetByName(SURVIVOR_SHEET_NAME); if (!sheet) return; const rows = sheet.getDataRange().getValues(); for (let i = 1; i < rows.length; i++) { if ((rows[i][0] || "").toString().trim() === playerName) { const rowNum = i + 1; // Weekly pick column: Player = column 1, week N -> column N+1. sheet.getRange(rowNum, Number(week) + 1).setValue(teamAbbr); // Teams Picked (column 22) - append if not already listed. const pickedCell = sheet.getRange(rowNum, 22);
